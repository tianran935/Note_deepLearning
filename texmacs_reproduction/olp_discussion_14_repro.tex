\[
\mathbb{E}\left[
\left|
\boxed{\color{red}
\sum_{t=1}^{T}\bigl(d^\star-a_t\mathbb{I}\{c_t\geq \langle a_t,y^t\rangle\},y^t\bigr)}
\right|
+
\left|
Td-\sum_{t=1}^{T}a_t\mathbb{I}\{c_t\geq \langle a_t,y^t\rangle\}
\right|
\right]
\]

Suppose \(y^t=y^\star\), then
\(\mathbb{E}[a_t\mathbb{I}\{c_t\geq \langle a_t,y^\star\rangle\}]=d\)
gives
\[
\mathbb{E}\!\left[
{\color{red}\sum_{t=1}^{T}
\bigl(d-a_t\mathbb{I}\{c_t\geq \langle a_t,y^t\rangle\},y^t\bigr)}
\right]=0.
\]

\begin{itemize}
\item[\textcolor{magenta}{\(\bullet\)}]
Constraint violation causes \(O(\sqrt{T})\) regret
\end{itemize}

Consider the constraint process starting from \(b_1=dT\)

\[
\begin{aligned}
a_t &= \frac{b_t}{T-t+1},\\
b_{t+1} &= b_t-a_tx^t .
\end{aligned}
\]

We have
\[
\mathbb{E}[\|d_t-d\|^2]
=O\!\left(\frac{t-1}{(T-t+1)^2}\right)
\quad\text{and}
\]

\[
\mathbb{E}[\|d_t-d\|]
=O\!\left(\frac{\sqrt{t-1}}{T-t+1}\right)
\]

\begin{itemize}
\item[\textcolor{magenta}{\(\bullet\)}]
\(\|d_t-d\|=\Omega(1)\) when
\(\sqrt{t-1}\sim T-t+1 \Rightarrow t=\Omega(T-\sqrt{T})\)
\end{itemize}
